## Descripción

Añadir soporte para reintentar un pipeline de compose desde un step fallido específico (`start_from_step: N`), reutilizando los outputs anteriores almacenados en DB. Incluye ownership check, cobro parcial desde step N, y select-for-update para concurrencia.

### 2.2 Flujo de retry (después de auth, antes del SSRF preflight)

```ts
// ── [2.5] RETRY MODE ─────────────────────────────────────────────────────
if (body.pipeline_id && body.start_from_step !== undefined) {
  const startFrom = body.start_from_step
  const retryPipelineId = body.pipeline_id

  // SELECT FOR UPDATE — previene concurrencia (AC-9)
  // Supabase no soporta FOR UPDATE nativo: usar RPC wrapper
  const { data: existingPipeline, error: pipelineErr } = await supabase
    .rpc('get_pipeline_for_retry', {
      p_pipeline_id: retryPipelineId,
      p_key_hash:    keyHash,
    })

  // Verificar que el pipeline existe, pertenece al key, y no está ya completado
  if (pipelineErr || !existingPipeline) {
    return NextResponse.json(
      { error: 'Pipeline not found or not resumable', code: 'pipeline_not_resumable' },
      { status: 404 }
    )
  }

  if (existingPipeline.status === 'success') {
    return NextResponse.json(
      { error: 'Pipeline already completed', code: 'pipeline_not_resumable' },
      { status: 409 }
    )
  }

  // Ownership check (AC-3, AC-4)
  if (!existingPipeline.owned_by_key) {
    return NextResponse.json(
      { error: 'Pipeline access denied', code: 'pipeline_access_denied' },
      { status: 403 }
    )
  }

  // Cargar outputs previos (AC-6)
  const previousOutputs: Array<{step: number; output: string}> = 
    existingPipeline.step_outputs ?? []
  
  // Override lastOutput si hay initial_input
  if (body.initial_input !== undefined) {
    lastOutput = body.initial_input
  } else if (previousOutputs.length > 0) {
    const lastStored = previousOutputs
      .filter(o => o.step < startFrom)
      .sort((a, b) => b.step - a.step)[0]
    lastOutput = lastStored?.output ?? null
  }

  // Ajustar globalStepIndex para cobrar solo desde N (AC-5)
  globalStepIndex = startFrom
  // Poblar receipts con placeholders de steps anteriores (no cobrados)
  for (let i = 0; i < startFrom; i++) {
    // No agregar a receipts — solo ajustar índice de inicio
  }

  // El response incluirá resumed_from_step (AC-7)
  resumedFromStep = startFrom
}
```

### 2.3 RPC requerida: `get_pipeline_for_retry`

```sql
-- En migration 052 también:
CREATE OR REPLACE FUNCTION get_pipeline_for_retry(
  p_pipeline_id UUID,
  p_key_hash    TEXT
)
RETURNS TABLE (
  id            UUID,
  status        TEXT,
  step_outputs  JSONB,
  owned_by_key  BOOLEAN
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  -- SELECT FOR UPDATE para lock concurrente
  RETURN QUERY
  SELECT
    pe.id,
    pe.status,
    pe.step_outputs,
    (ak.key_hash = p_key_hash) AS owned_by_key
  FROM pipeline_executions pe
  JOIN agent_keys ak ON ak.id = pe.key_id
  WHERE pe.id = p_pipeline_id
  FOR UPDATE;
END;
$$;

REVOKE EXECUTE ON FUNCTION get_pipeline_for_retry(UUID, TEXT) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION get_pipeline_for_retry(UUID, TEXT) TO service_role;
```

## Wave 3 — RPC append_step_output

**En migration 052:**

```sql
CREATE OR REPLACE FUNCTION append_step_output(
  p_pipeline_id UUID,
  p_step        INTEGER,
  p_output      TEXT,
  p_agent_slug  TEXT
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  UPDATE pipeline_executions
  SET step_outputs = COALESCE(step_outputs, '[]'::jsonb) || 
    jsonb_build_object('step', p_step, 'output', p_output, 'agent_slug', p_agent_slug)::jsonb
  WHERE id = p_pipeline_id;
END;
$$;

REVOKE EXECUTE ON FUNCTION append_step_output(UUID, INTEGER, TEXT, TEXT) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION append_step_output(UUID, INTEGER, TEXT, TEXT) TO service_role;
```

## Descripción

Permitir que un step de compose especifique `capability` (tipo de capacidad requerida) en lugar de `agent_slug`, delegando la resolución al servidor. El servidor elige el agente que mejor cumple los constraints, verifica scope de key (WAS-186), y devuelve el slug resuelto en el receipt.

## Wave 2 — Actualizar validateSteps

**Archivo:** `src/app/api/v1/compose/route.ts`

```ts
export function validateSteps(steps: unknown): string | null {
  if (!Array.isArray(steps))      return 'steps must be an array'
  if (steps.length < 1)           return 'steps must have at least 1 element'
  if (steps.length > MAX_STEPS)   return `Max ${MAX_STEPS} steps per pipeline`

  for (let i = 0; i < steps.length; i++) {
    const s = steps[i] as ComposeStep

    // AC-3: capability + agent_slug juntos → 400
    if (s.capability && s.agent_slug) {
      return `Step ${i}: capability and agent_slug are mutually exclusive`
    }

    // Debe tener uno u otro
    if (!s.capability && (!s.agent_slug || typeof s.agent_slug !== 'string')) {
      return `Step ${i}: agent_slug or capability is required`
    }

    if (s.input !== undefined && s.pass_output === true) {
      return `Step ${i}: input and pass_output are mutually exclusive`
    }
    if (i === 0 && s.pass_output === true) {
      return 'Step 0 cannot use pass_output (no previous output exists)'
    }
    if (!s.pass_output && (s.input === undefined || s.input.trim() === '')) {
      return `Step ${i}: input is required when pass_output is false`
    }
  }
  return null
}
```

## Wave 3 — Función de Discovery

**Archivo:** `src/lib/agent-discovery.ts` (nuevo)

```ts
import { SupabaseClient } from '@supabase/supabase-js'
import { isAgentInScope } from '@/lib/scope-check'  // WAS-186

interface DiscoveryConstraints {
  max_price_usdc?: number
  min_reputation?: number
  category?:       string
}

interface DiscoveredAgent {
  id:             string
  slug:           string
  name:           string
  category:       string
  price_per_call: number
  endpoint_url:   string
  status:         string
  max_rpm:        number
  max_rpd:        number
}

/**
 * Descubre el agente más adecuado para una capability dada.
 * Retorna null si no hay match.
 */
export async function discoverAgent(
  supabase:          SupabaseClient,
  capability:        string,
  constraints:       DiscoveryConstraints = {},
  allowedSlugs:      string[] | null,
  allowedCategories: string[] | null,
): Promise<DiscoveredAgent | null> {
  let query = supabase
    .from('agents')
    .select('id, slug, name, category, price_per_call, endpoint_url, status, max_rpm, max_rpd, capabilities, reputation_score')
    .eq('status', 'active')
    // JSON containment: capabilities @> '[{"name": "X"}]'
    .contains('capabilities', JSON.stringify([{ name: capability }]))

  if (constraints.max_price_usdc !== undefined) {
    query = query.lte('price_per_call', constraints.max_price_usdc)
  }
  if (constraints.min_reputation !== undefined) {
    query = query.gte('reputation_score', constraints.min_reputation)
  }
  if (constraints.category) {
    query = query.eq('category', constraints.category)
  }

  // Ordenar por reputación desc, precio asc
  const { data: candidates } = await query
    .order('reputation_score', { ascending: false })
    .order('price_per_call', { ascending: true })
    .limit(10)

  if (!candidates || candidates.length === 0) return null

  // AC-7: filtrar por scope de key (WAS-186 integration)
  const scopedCandidates = candidates.filter((a: DiscoveredAgent) =>
    isAgentInScope(a.slug, a.category, allowedSlugs, allowedCategories)
  )

  return scopedCandidates[0] ?? null
}
```

## Wave 4 — Integrar discovery en compose handler

**Archivo:** `src/app/api/v1/compose/route.ts`

### 4.1 Actualizar resolución de agentes (sección [3])

```ts
// ── [3] RESOLVER AGENTES ─────────────────────────────────────────────
// Separar steps con slug estático vs discovery dinámico
const staticSlugs = [...new Set(
  steps.filter(s => s.agent_slug).map(s => s.agent_slug!)
)]

let agentMap = new Map<string, AgentRow>()

if (staticSlugs.length > 0) {
  const { data: agentsData } = await supabase
    .from('agents')
    .select('id, slug, name, category, price_per_call, endpoint_url, status, max_rpm, max_rpd')
    .in('slug', staticSlugs)
    .eq('status', 'active')
  
  agentMap = new Map((agentsData ?? []).map((a: AgentRow) => [a.slug, a]))
}

// Resolver steps dinámicos por capability
const resolvedSlugs = new Map<number, string>() // stepIndex → resolved slug
for (let i = 0; i < steps.length; i++) {
  const step = steps[i]
  
  if (step.agent_slug) {
    // AC-3: agent_slug + capability → 400 (ya validado en validateSteps)
    if (!agentMap.has(step.agent_slug)) {
      return NextResponse.json(
        { error: 'Agent not found', code: 'agent_not_found', step: i, slug: step.agent_slug },
        { status: 404 }
      )
    }
    // Scope check (WAS-186)
    const agent = agentMap.get(step.agent_slug)!
    if (!isAgentInScope(agent.slug, agent.category, keyRow.allowed_slugs, keyRow.allowed_categories)) {
      return NextResponse.json(
        { error: 'Agent not in key scope', code: 'scope_violation', slug: agent.slug },
        { status: 403 }
      )
    }
  } else if (step.capability) {
    // Dynamic discovery
    const discovered = await discoverAgent(
      supabase,
      step.capability,
      step.constraints ?? {},
      keyRow.allowed_slugs,
      keyRow.allowed_categories,
    )

    if (!discovered) {
      // AC-6: intentar fallback_slug
      if (step.fallback_slug) {
        const fallback = agentMap.get(step.fallback_slug) 
          ?? (await supabase.from('agents').select('...').eq('slug', step.fallback_slug).single()).data
        if (fallback) {
          agentMap.set(step.fallback_slug, fallback as AgentRow)
          // Mutate step para usar fallback
          steps[i] = { ...step, agent_slug: step.fallback_slug }
          resolvedSlugs.set(i, step.fallback_slug)
          continue
        }
      }
      // AC-4: sin match → 422
      return NextResponse.json(
        { error: 'No agent found for capability', code: 'no_agent_match', step: i, capability: step.capability },
        { status: 422 }
      )
    }

    agentMap.set(discovered.slug, discovered as AgentRow)
    steps[i] = { ...step, agent_slug: discovered.slug }  // Mutar para resto del pipeline
    resolvedSlugs.set(i, discovered.slug)  // AC-5: tracking para receipt
  }
}
```

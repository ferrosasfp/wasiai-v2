## Descripción

Permitir a los creadores de agentes definir un JSON Schema (draft-07) para validar el input antes de ejecutar la llamada. La validación ocurre **antes del cobro**. Se bloquean referencias externas (`$ref` con URLs) para prevenir SSRF. La UI muestra el schema y ejemplos del agente.

**Archivo nuevo:** `src/lib/schema-validator.ts`

```ts
import Ajv from 'ajv'
import addFormats from 'ajv-formats'

const ajv = new Ajv({ strict: false })
addFormats(ajv)

/**
 * Meta-valida que un JSON Schema sea válido draft-07.
 * Bloquea $ref con URLs externas (SSRF prevention).
 */
export function metaValidateSchema(schema: unknown): { valid: boolean; error?: string } {
  // AC-3: Bloquear $ref con URL http/https (recursivo)
  const ssrfCheck = findExternalRefs(schema)
  if (ssrfCheck) {
    return { valid: false, error: ssrfCheck }
  }

  // AC-2: Meta-validar que sea JSON Schema draft-07 válido
  try {
    ajv.compile(schema)
    return { valid: true }
  } catch (e) {
    return { valid: false, error: e instanceof Error ? e.message : 'Invalid JSON Schema' }
  }
}

/**
 * Valida un input dado contra un JSON Schema.
 * Retorna null si válido, string de error si inválido.
 */
export function validateInput(schema: unknown, input: unknown): string | null {
  if (schema === null || schema === undefined) return null // AC-7: sin schema → pass

  const validate = ajv.compile(schema as object)
  const valid = validate(input)
  if (!valid) {
    return ajv.errorsText(validate.errors)
  }
  return null
}

/**
 * Busca recursivamente $ref con URLs externas.
 * Retorna mensaje de error o null si no hay SSRF.
 */
function findExternalRefs(obj: unknown, path = ''): string | null {
  if (typeof obj !== 'object' || obj === null) return null

  if (Array.isArray(obj)) {
    for (let i = 0; i < obj.length; i++) {
      const r = findExternalRefs(obj[i], `${path}[${i}]`)
      if (r) return r
    }
    return null
  }

  const record = obj as Record<string, unknown>
  for (const key of Object.keys(record)) {
    if (key === '$ref' && typeof record[key] === 'string') {
      const ref = record[key] as string
      if (ref.startsWith('http://') || ref.startsWith('https://')) {
        return `External $ref blocked at ${path}.$ref: ${ref}`
      }
    }
    const r = findExternalRefs(record[key], `${path}.${key}`)
    if (r) return r
  }
  return null
}
```

## Wave 3 — Register route: aceptar input_schema (AC-4)

**Archivo:** `src/app/api/v1/agents/register/route.ts`

### 3.1 Actualizar RegisterAgentSchema (Zod)

```ts
import { metaValidateSchema } from '@/lib/schema-validator'

const RegisterAgentSchema = z.object({
  // ...existing fields...
  input_schema: z.unknown().optional().nullable(),  // AC-4: acepta JSON Schema
})
.superRefine((data, ctx) => {
  if (data.input_schema !== undefined && data.input_schema !== null) {
    // AC-2: meta-validar
    const result = metaValidateSchema(data.input_schema)
    if (!result.valid) {
      ctx.addIssue({
        code: z.ZodIssueCode.custom,
        path: ['input_schema'],
        message: result.error ?? 'Invalid JSON Schema',
      })
    }
    // AC-3: SSRF check ya incluido en metaValidateSchema
  }
})
```

```ts
// En executeStep(), ANTES de deduct_key_balance:
if (agent.input_schema) {
  const validationError = validateInput(agent.input_schema, tryParseInput(stepInput))
  if (validationError) {
    // AC-6: sin cargo
    return { 
      receipt: null, output: null, status: 'error', 
      reason: validationError, 
      chargeDecision: 'refund',  // No se llegó a cobrar
      refundFailure: null 
    }
  }
}
```

**Nota:** El step retorna `status: 'error'` sin haber llamado `deduct_key_balance`, por lo que no hay refund necesario. El `chargeDecision: 'refund'` es por consistencia pero no tiene efecto.
